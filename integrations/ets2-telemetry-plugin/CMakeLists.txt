cmake_minimum_required(VERSION 3.20)
project(TruckRacingCareerTelemetry LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The ETS2 plugin currently targets Windows x64.")
endif()
if(NOT DEFINED SCS_SDK_INCLUDE_DIR OR NOT EXISTS "${SCS_SDK_INCLUDE_DIR}/scssdk_telemetry.h")
  message(FATAL_ERROR "Set SCS_SDK_INCLUDE_DIR to the official SCS SDK include directory.")
endif()
add_library(TruckRacingCareer SHARED TruckRacingCareer.cpp)
target_compile_features(TruckRacingCareer PRIVATE cxx_std_17)
target_include_directories(TruckRacingCareer PRIVATE "${SCS_SDK_INCLUDE_DIR}")
target_compile_definitions(TruckRacingCareer PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
set_target_properties(TruckRacingCareer PROPERTIES OUTPUT_NAME "TruckRacingCareer")
